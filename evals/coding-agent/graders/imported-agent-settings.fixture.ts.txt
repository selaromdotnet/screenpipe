// screenpipe — AI that knows everything you've seen, said, or heard
// https://screenpipe.com
import {beforeEach,expect,it,vi} from 'vitest';
const io=vi.hoisted(()=>({read:vi.fn(),stat:vi.fn(),load:vi.fn(),save:vi.fn(),emit:vi.fn()}));
vi.mock('@tauri-apps/api/event',()=>({emit:io.emit}));
vi.mock('@tauri-apps/api/path',()=>({homeDir:vi.fn(),join:vi.fn()}));
vi.mock('@tauri-apps/plugin-fs',()=>({exists:vi.fn(),readDir:vi.fn(),readTextFile:io.read,stat:io.stat}));
vi.mock('@/lib/chat-storage',()=>({deleteConversationFile:vi.fn(),loadConversationFile:io.load,saveConversationFile:io.save,invalidateConversationListCache:vi.fn()}));
import {importExternalChatHistory} from '@/lib/chat/external-chat-import';
const at=Date.parse('2026-08-01T12:00:00Z');
const context=(payload:any)=>({type:'turn_context',payload});
const user={type:'response_item',timestamp:at,payload:{type:'message',id:'u1',role:'user',content:[{type:'input_text',text:'Inspect the synthetic example'}]}};
function codex(extra:any[]=[]){return [{type:'session_meta',timestamp:at,payload:{id:'synthetic-session'}},...extra,user].map(x=>JSON.stringify(x)).join('\n');}
function claude(extra:any[]=[]){return [{type:'user',timestamp:at,sessionId:'synthetic-session',uuid:'u1',message:{role:'user',content:'Inspect the synthetic example'}},...extra].map(x=>JSON.stringify(x)).join('\n');}
async function run(source='codex',options={skipUnchanged:true}){return importExternalChatHistory([{source:source as any,path:'/fixture/synthetic.jsonl',sourceId:'synthetic-session',modifiedAt:at,size:1024}],options);}
function saved(){return io.save.mock.calls.at(-1)?.[0];}
beforeEach(()=>{vi.resetAllMocks();io.stat.mockResolvedValue({size:1024});io.load.mockResolvedValue(null);io.save.mockResolvedValue(undefined);io.emit.mockResolvedValue(undefined);});
it('saves latest portable Codex settings and excludes local authority metadata',async()=>{io.read.mockResolvedValue(codex([context({model:'synthetic-old',effort:'low'}),context({model:'synthetic-new',reasoning_effort:'high',cwd:'/private/example',env:{SYNTHETIC:'value'},approval_policy:'never',sandbox_policy:'unrestricted'})]));expect(await run()).toMatchObject({imported:1,failed:0});expect(saved().importedFrom.config).toEqual({model:'synthetic-new',reasoningEffort:'high'});});
it('saves Codex collaboration settings when direct settings are absent',async()=>{io.read.mockResolvedValue(codex([context({collaboration_mode:{settings:{model:'synthetic-plan',reasoning_effort:'medium'}}})]));await run();expect(saved().importedFrom.config).toEqual({model:'synthetic-plan',reasoningEffort:'medium'});});
it('saves Claude model and mode without taking synthetic or sidechain models',async()=>{io.read.mockResolvedValue(claude([{type:'assistant',timestamp:at+1,sessionId:'synthetic-session',permissionMode:'plan',message:{id:'a1',role:'assistant',model:'synthetic-claude',content:[{type:'text',text:'A synthetic response'}]}},{type:'assistant',timestamp:at+2,sessionId:'synthetic-session',message:{id:'a2',role:'assistant',model:'<synthetic>',content:[{type:'text',text:'Synthetic wrapper'}]}},{type:'assistant',timestamp:at+3,isSidechain:true,sessionId:'synthetic-session',message:{id:'other',role:'assistant',model:'wrong-sidechain',content:[{type:'text',text:'Sidechain'}]}}]));await run('claude-code');expect(saved().importedFrom.config).toEqual({model:'synthetic-claude',modeId:'plan'});});
it('persists a settings-only refresh even when message content and timestamps match',async()=>{io.read.mockResolvedValue(codex([context({model:'synthetic-new',effort:'high'})]));await run();const existing=structuredClone(saved());existing.importedFrom.config={model:'synthetic-old',reasoningEffort:'low'};io.load.mockResolvedValue(existing);io.save.mockClear();expect(await run()).toMatchObject({updated:1,skipped:0,failed:0});expect(saved().importedFrom.config).toEqual({model:'synthetic-new',reasoningEffort:'high'});});
it('preserves manual title, selected preset, pin and local continuation',async()=>{io.read.mockResolvedValue(codex());await run();const existing=structuredClone(saved());Object.assign(existing,{title:'My chosen title',titleSource:'user',presetId:'chosen-preset',pinned:true});existing.messages.push({id:'local-reply',role:'assistant',content:'Local continuation',timestamp:at+100});io.load.mockResolvedValue(existing);io.save.mockClear();io.read.mockResolvedValue(codex([context({model:'synthetic-new'})]));await run('codex',{skipUnchanged:false});expect(saved()).toMatchObject({title:'My chosen title',titleSource:'user',presetId:'chosen-preset',pinned:true});expect(saved().messages.at(-1)).toMatchObject({id:'local-reply',content:'Local continuation'});});
it('does not invent settings for a transcript without configuration',async()=>{io.read.mockResolvedValue(codex());await run();expect(saved().importedFrom.config).toBeUndefined();expect(saved().messages[0].content).toBe('Inspect the synthetic example');});
it('skips fully unchanged records without rewriting them',async()=>{io.read.mockResolvedValue(codex());await run();io.load.mockResolvedValue(structuredClone(saved()));io.save.mockClear();expect(await run()).toMatchObject({updated:0,skipped:1,failed:0});expect(io.save).not.toHaveBeenCalled();});
it('keeps an unreadable input out of storage and reports failure',async()=>{io.read.mockRejectedValue(new Error('synthetic unavailable input'));expect(await run()).toMatchObject({imported:0,updated:0,failed:1});expect(io.save).not.toHaveBeenCalled();});
