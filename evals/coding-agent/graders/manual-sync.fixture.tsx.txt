function mount() {vi.stubGlobal("fetch",vi.fn(async () => {throw Error("unexpected external request")})); return render(<AccountSection/>);}
function button(index: number) {
 const labels=[/^(pipe sync|sync scheduled tasks) across devices$/i,/^memories sync across devices$/i,/^connection sync across devices$/i];
 const label=screen.getByText(labels[index]);let group=label.parentElement;
 while(group && !within(group).queryByRole("button",{name:/^sync now$/i}))group=group.parentElement;
 expect(group).not.toBeNull();return within(group!).getByRole("button",{name:/^sync now$/i});
}
const targets=["pipes","memories","connections"];
function reply(status:number, body:any) {return new Response(JSON.stringify(body),{status,headers:{"Content-Type":"application/json"}});}
for (const [index,target] of targets.entries()) {
 it(target+" pull rejection prevents push and success",async()=>{
  mocks.fetch.mockResolvedValue(reply(400,{error:"synthetic sync refused"}));mount();fireEvent.click(button(index));
  await waitFor(()=>expect(mocks.toast).toHaveBeenCalledTimes(1));
  expect(mocks.toast.mock.calls[0][0]).toMatchObject({variant:"destructive",description:expect.stringContaining("synthetic sync refused")});
  expect(mocks.fetch.mock.calls.map(x=>x[0])).toEqual(["/sync/"+target+"/pull"]);expect(button(index)).not.toBeDisabled();
 });
 it(target+" push rejection is failure after successful pull",async()=>{
  mocks.fetch.mockResolvedValueOnce(reply(200,{})).mockResolvedValueOnce(reply(401,{error:"synthetic unauthorized"}));mount();fireEvent.click(button(index));
  await waitFor(()=>expect(mocks.toast).toHaveBeenCalledTimes(1));expect(mocks.toast.mock.calls[0][0]).toMatchObject({variant:"destructive",description:expect.stringContaining("synthetic unauthorized")});
  expect(mocks.fetch.mock.calls.map(x=>x[0])).toEqual(["/sync/"+target+"/pull","/sync/"+target+"/push"]);expect(button(index)).not.toBeDisabled();
 });
 it(target+" preserves successful pull then push",async()=>{
  mocks.fetch.mockResolvedValue(reply(200,{}));mount();fireEvent.click(button(index));
  await waitFor(()=>expect(mocks.toast).toHaveBeenCalledTimes(1));expect(mocks.toast.mock.calls[0][0].variant).not.toBe("destructive");expect(mocks.toast.mock.calls[0][0].title).toMatch(/synced/i);
  expect(mocks.fetch.mock.calls.map(x=>[x[0],x[1].method])).toEqual([["/sync/"+target+"/pull","POST"],["/sync/"+target+"/push","POST"]]);expect(button(index)).not.toBeDisabled();
 });
 it(target+" handles non JSON response and retries",async()=>{
  mocks.fetch.mockResolvedValueOnce(new Response("synthetic unavailable",{status:503}));mount();fireEvent.click(button(index));
  await waitFor(()=>expect(mocks.toast).toHaveBeenCalledTimes(1));expect(mocks.toast.mock.calls[0][0]).toMatchObject({variant:"destructive",description:expect.stringContaining("503")});
  mocks.fetch.mockResolvedValue(reply(200,{}));fireEvent.click(button(index));await waitFor(()=>expect(mocks.toast).toHaveBeenCalledTimes(2));expect(mocks.toast.mock.calls[1][0].title).toMatch(/synced/i);expect(mocks.fetch).toHaveBeenCalledTimes(3);
 });
 it(target+" preserves network failure feedback",async()=>{
  mocks.fetch.mockRejectedValue(new TypeError("Load failed"));mount();fireEvent.click(button(index));await waitFor(()=>expect(mocks.toast).toHaveBeenCalledTimes(1));expect(mocks.toast.mock.calls[0][0]).toMatchObject({variant:"destructive"});expect(String(mocks.toast.mock.calls[0][0].description)).toMatch(/server.*reachable/i);expect(mocks.fetch).toHaveBeenCalledTimes(1);expect(button(index)).not.toBeDisabled();
 });
}
it("disabled sync choices do not expose manual actions or send requests",()=>{mocks.state.enabled=false;mount();expect(screen.queryAllByRole("button",{name:/^sync now$/i})).toHaveLength(0);expect(mocks.fetch).not.toHaveBeenCalled();});
